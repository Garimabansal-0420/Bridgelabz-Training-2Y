import java.util.*;

class Participant<T> {
    private T rollNumber;
    private String name;
    private String branch;

    public Participant(T rollNumber, String name, String branch) {
        this.rollNumber = rollNumber;
        this.name = name;
        this.branch = branch;
    }

    public T getRollNumber() {
        return rollNumber;
    }

    public String getName() {
        return name;
    }

    public String getBranch() {
        return branch;
    }

    @Override
    public String toString() {
        return "Roll No: " + rollNumber + ", Name: " + name + ", Branch: " + branch;
    }
}

public class EventRegistrationSystem {
    private static final int MAX_REGISTRATIONS = 3; // Max participants allowed

    private ArrayList<Participant<Integer>> registeredList = new ArrayList<>();
    private LinkedList<Participant<Integer>> waitingList = new LinkedList<>();
    private HashSet<Integer> registeredRollNumbers = new HashSet<>();
    
    public void registerParticipant(Participant<Integer> p) {
        if (registeredRollNumbers.contains(p.getRollNumber())) {
            System.out.print("Duplicate roll number! Registration denied for: " + p.getName());
            return;
        }

        if (registeredList.size() < MAX_REGISTRATIONS) {
            registeredList.add(p);
            registeredRollNumbers.add(p.getRollNumber());
            System.out.println("✅ Registered: " + p.getName());
        } else {
            waitingList.add(p);
            registeredRollNumbers.add(p.getRollNumber());
            System.out.println(" Added to waiting list: " + p.getName());
        }
    }
    
    public void cancelRegistration(int rollNumber) {
        boolean removed = false;
        Iterator<Participant<Integer>> iterator = registeredList.iterator();

        while (iterator.hasNext()) {
            Participant<Integer> p = iterator.next();
            if (p.getRollNumber() == rollNumber) {
                iterator.remove();
                registeredRollNumbers.remove(p.getRollNumber());
                removed = true;
               System.out.print("Cancelled registration for: " + p.getName());
               break;
            }
        }

        if (removed && !waitingList.isEmpty()) {
            Participant<Integer> next = waitingList.poll();
            registeredList.add(next);
            System.out.println("➡  Moved from waiting list to registered: " + next.getName());
        }
    }

    public void displayParticipants() {
        System.out.println("\n===== Registered Participants =====");
        for (Participant<Integer> p : registeredList) {
            System.out.println(p);
        }

        System.out.println("\n===== Waiting List =====");
        for (Participant<Integer> p : waitingList) {
            System.out.println(p);
        }
        System.out.println();
    }

    public static void main(String[] args) {
        EventRegistrationSystem system = new EventRegistrationSystem();

        Participant<Integer> p1 = new Participant<>(101, "Garima", "CSE");
        Participant<Integer> p2 = new Participant<>(102, "isha", "ECE");
        Participant<Integer> p3 = new Participant<>(103, "aditya", "ME");
        Participant<Integer> p4 = new Participant<>(104, "ishaan", "CSE");
        Participant<Integer> p5 = new Participant<>(102, "palak", "IT"); // Duplicate roll number

        system.registerParticipant(p1);
        system.registerParticipant(p2);
        system.registerParticipant(p3);
        system.registerParticipant(p4);
        system.registerParticipant(p5);

        system.displayParticipants();

        system.cancelRegistration(102);

        system.displayParticipants();
    }
}
